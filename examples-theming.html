CSS Custom Properties Reference

/* Core colors */
--ltree-primary: #059669;          /* Accent/primary color */
--ltree-text: #1e293b;             /* Default text color */
--ltree-bg: transparent;           /* Tree background */

/* Interaction states */
--ltree-hover-bg: #f0fdf4;         /* Node hover background */
--ltree-selected-bg: #065f46;      /* Selected node background */
--ltree-selected-text: #dcfce7;    /* Selected node text color */

/* Icons */
--ltree-icon-color: #059669;       /* Toggle icon color */
--ltree-icon-size: 0.8em;          /* Toggle icon size */

/* Layout */
--ltree-indent-size: 20px;         /* Per-level indentation */
--ltree-node-height: auto;         /* Node row height */
--ltree-node-padding: 4px 8px;     /* Node content padding */

/* Borders & misc */
--ltree-border-color: #e2e8f0;     /* Border color */
--ltree-border-radius: 4px;        /* Node border radius */
